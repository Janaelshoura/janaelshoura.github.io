TRIP DISTANCE USING HAVERSINE FORMULA

Trip distance was estimated using the Haversine formula, which calculates the geographical distance between two latitude and longitude coordinates on Earth.

from pyspark.sql.functions import radians, sin, cos, sqrt, atan2

R = 6371  # Earth radius in KM

df = df.withColumn(
    "dlat",
    radians(col("end station latitude") - col("start station latitude"))
)

df = df.withColumn(
    "dlon",
    radians(col("end station longitude") - col("start station longitude"))
)

df = df.withColumn(
    "a",
    sin(col("dlat") / 2) ** 2 +
    cos(radians(col("start station latitude"))) *
    cos(radians(col("end station latitude"))) *
    sin(col("dlon") / 2) ** 2
)

df = df.withColumn(
    "c",
    2 * atan2(sqrt(col("a")), sqrt(1 - col("a")))
)

df = df.withColumn(
    "Trip_Distance",
    col("c") * R
)

NOISE DETECTION

Noise detection was performed to identify unrealistic or invalid records that could negatively affect analysis accuracy and machine learning performance.

df = df.withColumn(
    "Noise_Flag",

    when(col("Trip_Duration") < 60,
         "Short Trip")

    .when(col("Trip_Speed") > 40,
          "Unrealistic Speed")

    .when((col("Age") > 100) | (col("Age") < 12),
          "Invalid Age")

    .when(
        col("start station name").isNull() |
        col("end station name").isNull() |
        col("bikeid").isNull(),
        "Missing Essential Data"
    )

    .otherwise("Valid")
)

def season(month):
